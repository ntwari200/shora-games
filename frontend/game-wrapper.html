// ============================================================ // FULL DESKTOP SITE TOGGLE (Chrome-like) // ============================================================ function toggleDesktopMode() { const params = new URLSearchParams(window.location.search); const isDesktop = params.get('desktop') === 'true'; if (isDesktop) { // Switch to mobile params.delete('desktop'); // Change viewport document.querySelector('meta[name=viewport]').content = 'width=device-width, initial-scale=1.0, maximum-scale=1.0, user-scalable=no'; // Reload with mobile settings window.location.search = params.toString(); } else { // Switch to desktop params.set('desktop', 'true'); // Change viewport to desktop width (980px = Chrome default) document.querySelector('meta[name=viewport]').content = 'width=980, initial-scale=1.0'; // Reload with desktop settings window.location.search = params.toString(); } }
